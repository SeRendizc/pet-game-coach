% 小芽项目报告 —— 排版设置
% 编译：tectonic（XeTeX 引擎）+ ctex
%
% 注意：
% 1. 不设 \XeTeXlinebreaklocale / \XeTeXlinebreakskip。断行由 xeCJK 的 \CJKglue 处理，
%    手设这两个裸 primitive 在 ctex 下是不必要的。
% 2. 不用 titlesec。加载它会完全接管章节标题，ctex 自带的标题间距设置会失效。
%    章节样式改用 ctex 自己的 \ctexset。
% 3. 不用 \sloppy。它的作用是放宽断行阈值，代价是把不良断行的警告一起盖掉，
%    等于把问题藏起来。真正该做的是让长串可断（\path、表格用 p{} 列）并把内容写短。
% 4. 不写 \setlength\parindent。ctex 默认 autoindent=true，已经等于 2\ccwd，
%    而 2\ccwd 包含字距，与 2em 并不等价，手写反而容易错。
% 5. \parskip 设为固定值而不是含 stretch 的胶：正文段落之间要能喘气，
%    但可伸长的段间距会把页断行的松紧吃进去，反而更容易出 underfull \vbox。
\documentclass[UTF8, scheme=chinese, a4paper, zihao=-4]{ctexart}

\usepackage{geometry}
\geometry{a4paper, top=2.4cm, bottom=2.4cm, left=2.5cm, right=2.5cm}

\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{array}
\usepackage{enumitem}
\usepackage{xurl}              % 允许 URL 任意位置断行
\usepackage{tikz}              % 只用来画第四节的执行链框图；不引 decorations 这类可选库
\usetikzlibrary{arrows.meta,positioning}
\usepackage[hidelinks]{hyperref}

\linespread{1.16}
\setlength{\parskip}{0.45em}

% 章节样式：用 ctex 自己的接口，不引 titlesec
\ctexset{
  section/format    = \normalfont\large\bfseries,
  subsection/format = \normalfont\normalsize\bfseries,
  section/beforeskip  = 1.3em,
  section/afterskip   = 0.5em,
  subsection/beforeskip = 0.9em,
  subsection/afterskip  = 0.35em,
}

% 浮动体间距：默认 12pt 偏松，报告里图表多，收紧能省出小半页
\setlength{\intextsep}{6pt plus 1pt minus 1pt}
\setlength{\textfloatsep}{6pt plus 1pt minus 1pt}

\pagestyle{plain}

\newcommand{\code}[1]{\texttt{\small #1}}
\newcommand{\file}[1]{\texttt{\small #1}}

% 定宽表格列：内容短，不设 \sloppy 也不会溢出
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}

% 截图：每张图带编号与一行短标题，图注写在正文里，正文用 \ref 引用编号。
%
% 宽度仍按「读得清」定，不按「省地方」定：本报告正文栏宽 453pt，一个整栏宽的界面元素
% 在 1000 CSS px 视口下有 956 px，缩到 45% 栏宽时 12px 的依据正文只剩 2.9pt——
% 那种图放进 PDF 等于没有。用浮动体而不是行内 center：行内的整栏大图一旦在本页放不下，
% 就会把整页剩下的空白一起带走（第一版 9 张图把报告从 9 页顶到 12 页，多出来的几乎全是空隙）。
%
% 编号规则：**一个 figure 环境 = 一个编号**。\shotpair 并排的两张算一张图，
% 用（a）（b）标明哪张是哪块界面——它们是同一次操作的两半，拆成两个编号反而要多写一句
% 「图 5 与图 6 是同一次出征」。\shot 与 \shotpair 共用同一个 shot 计数器，所以
% 正文里的「图 3」在两种图混排时仍然连续，引用一律走 \ref{fig:<文件名>}，不手写数字。
%
% 编号写在图下的一行小字里，没有题注框、不加 caption 宏包：本报告的字号、行距与浮动体
% 间距都是手调的，caption 会再带进一套默认间距与「Figure」样式。
% \parbox 是给标题留的固定宽度容器，标题长了会自己折行，不会顶出 overfull。
\newcounter{shot}
\newcommand{\shotcaption}[1]{%
  \par\vspace{3pt}%
  \refstepcounter{shot}%
  \parbox[t]{0.96\linewidth}{\centering\small 图~\theshot\quad #1}\par}

% 单张截图。原来只给图不给标题，图注写在正文里，读者在图上找不到它对应正文哪一句。
\newcommand{\shot}[3][0.72]{%
  \begin{figure}[htbp]\centering
    \includegraphics[width=#1\linewidth]{figs/#2}%
    \shotcaption{#3}\label{fig:#2}%
  \end{figure}}

% 并排两张截图：各配一行小字说明「哪张是哪块界面」，下面一行是这一张图的编号与短标题。
% 两块加起来 0.92 栏宽，剩下的留给 \hfill，不会 overfull。
% 开头的 \vspace{0pt} 不是装饰：minipage 的 [t] 对齐的是**第一行的基线**，
% 而图片那一行的基线在图片底边，两张不一样高的图于是变成底对齐，一高一低。
% 加一个零高的空行把基线提到顶端，才是真的顶对齐。
\newcommand{\shotpair}[6][0.46]{%
  \begin{figure}[htbp]\centering
  \begin{minipage}[t]{#1\linewidth}\vspace{0pt}\centering
    \includegraphics[width=\linewidth]{figs/#2}\\[3pt]{\small（a）#3}%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{#1\linewidth}\vspace{0pt}\centering
    \includegraphics[width=\linewidth]{figs/#4}\\[3pt]{\small（b）#5}%
  \end{minipage}%
  \shotcaption{#6}\label{fig:#2}%
  \end{figure}}
